\section{Ground Rules}
\begin{itemize}
    \item Momentum, $p$, is conserved in isolated systems.
    \item Kinetic energy, $K$, is conserved in elastic collisions.
    \item Kinetic energy $K$, is not conserved in inelastic collisions.
    \item $p = mv$
    \item $K = \tfrac{1}{2}mv^2$
\end{itemize}
