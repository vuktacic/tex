\section{Materials}
\begin{itemize}
    \item 0.010M \ce{Pb(NO3)2} solution
    \item 0.020M \ce{KI} solution
    \item Erlenmeyer flask ($>$100mL)
    \item Graduated cylinders (10mL \& 50mL)
    \item Burette
\end{itemize}

\subsection{Alternative Materials}
\begin{itemize}
    \item 0.010M \ce{Pb(NO3)2} solution
    \item 0.020M \ce{KI} solution
    \item Beaker ($>$100mL)
    \item Hot air oven
    \item Filter paper
\end{itemize}

\section{Procedure}
\begin{itemize}
    \item Dilute 10.0mL of 0.010M \ce{Pb(NO3)2} solution to 50.0mL (0.0020M) with distilled water. Transfer to the Erlenmeyer flask.
    \item Pour 0.020M \ce{KI} solution into the burette
    \item Titrate the \ce{KI} solution into the \ce{Pb(NO3)2} solution until a precipitate forms.
    \item Record $\Delta V$ of \ce{KI}. This will be used to calculate \ksp.
\end{itemize}

\subsection{Alternative Procedure}
\begin{itemize}
    \item Mix 10.0mL of 0.010M \ce{Pb(NO3)2} solution with 10.0mL of 0.020M \ce{KI} solution in a beaker.
    \item Let mixture sit to allow precipitate to form.
    \item Filter precipitate out of solution and dry in hot air oven
    \item Measure mass of dried precipitate. This will be used to calculate \ksp.
\end{itemize}
